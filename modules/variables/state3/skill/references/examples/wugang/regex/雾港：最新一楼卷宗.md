(() => {
  'use strict';
  const copy = {
    host: '无法连接当前聊天。请在酒馆聊天中打开卷宗。',
    missing: '尚未读取到案卷。请在本聊天启用变量 3.0，或使用原生聊天控制恢复案卷。',
    invalid: path => '案卷内容不完整或格式错误，暂不展示。请检查本聊天的案卷：' + path,
    read: detail => '读取案卷失败，暂不展示。原因：' + detail,
    changed: '聊天已切换，此卷宗已停止读取。请查看新聊天最新一楼的卷宗。',
    clock: '审理时限', elapsed: (minute, limit) => minute + ' / ' + limit + ' 分钟',
    remaining: minutes => minutes ? '距截止 ' + minutes + ' 分钟' : '审理时限已到',
    exhibits: '证物册', people: '在场人员', leads: '已发现线索', claims: '证言记录', verdict: '审理结论',
    source: value => '来源：' + value, custodian: value => '保管：' + value, uncollected: '尚未取得',
    authenticity: { unknown: '真伪待定', genuine: '真品', forged: '伪造' },
    contamination: { clean: '未污染', tainted: '已污染', destroyed: '已毁损' },
    chain: ['保管链已断', '保管链完整'], seal: ['未封存', '已封存'], destroyed: '已销毁',
    inspection: ['未检视', '已检视'], tested: { none: '未得检验结论', match: '检验吻合', mismatch: '检验不吻合' },
    trust: value => '配合度 ' + value + ' / 10', testified: ['未作证', '已作证'],
    noLeads: '尚未发现线索；继续在聊天中调查。', noClaims: '尚未录入证言。',
    topic: value => '主题：' + value, stance: value => '指向：' + value,
    accused: value => '指控对象：' + value, evidence: value => '采用证物：' + value,
  };
  const current = document.getElementById('current'), errorBox = document.getElementById('error');
  let chatId, previous, disposed = false, timer;
  const fault = (code, message) => Object.assign(new Error(message), { code });
  function el(tag, value, className) {
    const node = document.createElement(tag);
    if (value !== undefined) node.textContent = value;
    if (className) node.className = className;
    return node;
  }
  function section(id, title) {
    const node = el('section');
    node.id = id;
    node.setAttribute('aria-labelledby', id + '-title');
    const heading = el('h2', title);
    heading.id = id + '-title';
    node.append(heading);
    return node;
  }
  function tag(value, tone) {
    const node = el('span', value, 'tag');
    if (tone) node.dataset.tone = tone;
    return node;
  }
  function render(d) {
    const name = code => Object.hasOwn(d.people, code) ? d.people[code].name : code;
    const overview = section('overview', copy.clock), clock = el('div', undefined, 'clock');
    const remaining = d.clock.limit - d.clock.minute;
    clock.dataset.urgency = remaining === 0 ? 'out' : remaining <= 60 ? 'low' : 'normal';
    clock.append(el('strong', copy.elapsed(d.clock.minute, d.clock.limit)), el('span', copy.remaining(remaining), 'remaining'));
    overview.append(clock, el('p', d.note, 'note'));

    const exhibits = section('exhibits', copy.exhibits), cards = el('div', undefined, 'cards');
    for (const e of Object.values(d.exhibits)) {
      const card = el('article', undefined, 'record'), header = el('header');
      card.dataset.exhibit = e.id;
      card.dataset.collected = String(e.custodian !== '');
      header.append(el('span', e.id, 'id'), el('h3', e.name));
      card.append(header, el('p', copy.source(e.source), 'meta'), el('p', e.custodian ? copy.custodian(name(e.custodian)) : copy.uncollected, 'meta'));
      const tags = el('div', undefined, 'tags');
      tags.append(tag(copy.authenticity[e.authenticity], e.authenticity === 'genuine' ? 'good' : e.authenticity === 'forged' ? 'bad' : ''),
        tag(copy.contamination[e.contamination], e.contamination === 'clean' ? '' : 'bad'),
        tag(copy.chain[Number(e.legalChain)], e.legalChain ? '' : 'bad'), tag(copy.seal[Number(e.sealed)], e.sealed ? 'good' : ''),
        tag(copy.inspection[Number(e.inspected)]), tag(copy.tested[e.tested], e.tested === 'match' ? 'good' : e.tested === 'mismatch' ? 'bad' : ''));
      if (e.destroyed) tags.append(tag(copy.destroyed, 'bad'));
      card.append(tags, el('p', e.testLabel, 'meta'));
      cards.append(card);
    }
    exhibits.append(cards);

    const people = section('people', copy.people), personList = el('ul', undefined, 'people');
    for (const [id, p] of Object.entries(d.people)) {
      const row = el('li', undefined, 'person'), identity = el('div'), trust = el('div', undefined, 'trust');
      row.dataset.person = id;
      row.dataset.trustBand = p.trust >= 6 ? 'cooperative' : p.trust >= 4 ? 'cautious' : 'guarded';
      identity.append(el('h3', p.name), el('p', p.role, 'meta'));
      trust.append(el('p', copy.trust(p.trust)), el('p', copy.testified[Number(p.testified)], 'meta'));
      row.append(identity, trust);
      personList.append(row);
    }
    people.append(personList);

    const leads = section('leads', copy.leads), leadList = el('ul', undefined, 'leads');
    for (const lead of Object.values(d.leads)) {
      if (!lead.opened) continue;
      const row = el('li');
      row.dataset.lead = lead.id;
      row.append(el('span', lead.id + ' · ', 'id'), el('span', lead.title));
      leadList.append(row);
    }
    leads.append(leadList.childElementCount ? leadList : el('p', copy.noLeads, 'empty'));

    const claims = section('claims', copy.claims), claimList = el('ul', undefined, 'claims');
    for (const c of Object.values(d.claims)) {
      const row = el('li', undefined, 'record'), header = el('header');
      row.dataset.claim = c.id;
      header.append(el('span', c.id, 'id'), el('h3', name(c.speaker)));
      row.append(header, el('div', copy.topic(c.topic), 'meta'));
      if (c.stance) row.append(el('div', copy.stance(name(c.stance)), 'meta'));
      row.append(el('p', c.text));
      claimList.append(row);
    }
    claims.append(claimList.childElementCount ? claimList : el('p', copy.noClaims, 'empty'));

    const verdict = section('verdict', copy.verdict), result = el('div', undefined, 'verdict');
    result.dataset.status = d.verdict.status;
    result.append(el('h3', d.verdict.label));
    if (d.verdict.accused !== null) result.append(el('p', copy.accused(name(d.verdict.accused))));
    if (d.verdict.evidenceUsed.length) result.append(el('p', copy.evidence(d.verdict.evidenceUsed.map(id => id + ' ' + d.exhibits[id].name).join('、')), 'meta'));
    if (d.verdict.reason) result.append(el('p', d.verdict.reason));
    verdict.append(result);
    current.replaceChildren(overview, exhibits, people, leads, claims, verdict);
    errorBox.hidden = true;
    errorBox.textContent = '';
    delete errorBox.dataset.error;
  }
  function showError(code, message) {
    previous = undefined;
    current.replaceChildren();
    errorBox.dataset.error = code;
    errorBox.textContent = message;
    errorBox.hidden = false;
  }
  function dispose() {
    if (disposed) return;
    disposed = true;
    window.clearInterval(timer);
    window.removeEventListener('pagehide', dispose);
  }
  function refresh() {
    if (disposed) return;
    try {
      const host = window.parent;
      if (!host || !host.SillyTavern || typeof host.SillyTavern.getContext !== 'function') throw fault('host-unavailable', copy.host);
      const ctx = host.SillyTavern.getContext();
      if (!ctx || ctx.chatId === undefined || ctx.chatId === null || ctx.chatId === '') throw fault('host-unavailable', copy.host);
      if (chatId === undefined) chatId = ctx.chatId;
      if (chatId !== ctx.chatId) {
        showError('chat-changed', copy.changed);
        dispose();
        return;
      }
      if (!ctx.variables || !ctx.variables.local || typeof ctx.variables.local.get !== 'function') throw fault('host-unavailable', copy.host);
      const raw = ctx.variables.local.get('docket');
      if (raw === undefined || raw === null || raw === '') throw fault('missing-state', copy.missing);
      if (raw === previous) return;
      let data;
      try { data = JSON.parse(raw); }
      catch (error) { throw fault('invalid-state', copy.invalid(error.message)); }
      if (data === null || typeof data !== 'object' || Array.isArray(data)) throw fault('invalid-state', copy.invalid('docket'));
      try { render(data); }
      catch (error) { throw fault('invalid-state', copy.invalid(error.message)); }
      previous = raw;
    } catch (error) {
      showError(error.code || 'read-failed', error.code ? error.message : copy.read(error.message || String(error)));
    }
  }
  window.addEventListener('pagehide', dispose, { once: true });
  refresh();
  if (!disposed) timer = window.setInterval(refresh, 500);
})();
</script>
