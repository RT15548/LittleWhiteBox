  function reasonOf(res) {
    var list = (res && res.diagnostics) || [];
    var d = list[0] || res || {};
    var cond = d.condition || {};
    if (cond.left && typeof cond.left.value === 'string' && cond.left.value) return cond.left.value;

  function toast(msg, bad) {
    var t = $('toast');
    t.textContent = msg;
    t.className = 'toast' + (bad ? ' bad' : '');

  // 一次提交：等结算完再返回。调用方负责 busy 锁。失败不改任何值；保存未确认时只提供重试保存，不重发。
  async function settle(ops) {
    if (S.gen) throw new Error(COPY.generating);
    var api = stateApi();
    if (!api) throw new Error(COPY.noApi);
    var res = await api.apply(ops);
    if (!res.ok) throw new Error(reasonOf(res) || COPY.failed);
    S.retry = res.saved === false ? res.receipt && res.receipt.id : null;
    return res;
  }
  async function run(ops, done) {
    if (S.busy) return;
    S.busy = true; render();
    var fundsBefore = S.club.funds;
    try {
      var res = await settle(ops);
      readState();
      var msg = done ? done(res) : '';
      var delta = S.club.funds - fundsBefore;
      if (delta) msg += '　账上 ' + signed(delta) + ' 元';
      if (msg) { S.notice = msg; toast(msg + (S.retry ? '\n' + COPY.unsaved : ''), !!S.retry); }
      else if (S.retry) toast(COPY.unsaved, true);
    } catch (e) {
      readState();
      toast(COPY.failed + '：' + e.message, true);
    } finally {
      S.busy = false; S.confirm = null; render();
    }
  }
  async function retrySave() {
    var api = stateApi();
    if (!api || S.busy) return;
    S.busy = true; render();
    try { await api.retrySave(); S.retry = null; toast(COPY.saved); }
    catch (e) { toast(COPY.failed + '：' + e.message, true); }
    finally { S.busy = false; readState(); render(); }
  }

  // ---- 线：一个人的失控画成一根线。越失控，绕得越多、越紧；底下是铃，失控降到结业线下时亮。
  function seeded(str) {
    var h = 2166136261;
    for (var i = 0; i < str.length; i++) { h ^= str.charCodeAt(i); h = Math.imul(h, 16777619); }
    return function () { h = Math.imul(h ^ (h >>> 15), 2246822507); h = Math.imul(h ^ (h >>> 13), 3266489909); return ((h ^= h >>> 16) >>> 0) / 4294967296; };
  }
  // 线 = 一条从左到铃的直线，与一团乱麻按权重混合：中段权重最高，乱麻的大小、圈数随失控增长。失控低时几乎是一根松弛的直线。
  function threadPath(name, loss, W, H, x0) {
    var rnd = seeded(name), k = Math.max(0, Math.min(1, loss / 100)), mid = H / 2, len = W - x0;
    var cx = x0 + len * (0.42 + rnd() * 0.16), Rx = 3 + k * Math.min(len * 0.22, H * 1.15), Ry = 1 + k * H * 0.44;
    var turns = 0.4 + k * k * 9, f = [1 + rnd() * 1.6, 2.3 + rnd() * 2, 0.8 + rnd() * 1.4, 2 + rnd() * 2.2], p = [rnd() * 6.28, rnd() * 6.28, rnd() * 6.28, rnd() * 6.28];
    var n = 60 + Math.round(k * 260), d = '', prev = null;
    for (var i = 0; i <= n; i++) {
      var s = i / n, w = Math.pow(Math.sin(Math.PI * s), 0.55), a = s * turns * 6.283;
      var tx = cx + Rx * (0.62 * Math.sin(f[0] * a + p[0]) + 0.38 * Math.sin(f[1] * a + p[1]));
      var ty = mid + Ry * (0.62 * Math.cos(f[2] * a + p[2]) + 0.38 * Math.cos(f[3] * a + p[3]));
      var lx = x0 + s * len, ly = mid + Math.sin(s * Math.PI) * (1 - k) * 2.5;
      var mix = Math.min(1, w * (0.15 + k * 1.1)), x = lx + (tx - lx) * mix, y = ly + (ty - ly) * mix;
      if (!prev) d = 'M' + x.toFixed(1) + ' ' + y.toFixed(1);
      else d += 'Q' + prev[0].toFixed(1) + ' ' + prev[1].toFixed(1) + ' ' + ((prev[0] + x) / 2).toFixed(1) + ' ' + ((prev[1] + y) / 2).toFixed(1);
      prev = [x, y];
    }
    return d + 'L' + prev[0].toFixed(1) + ' ' + prev[1].toFixed(1);

  // 两人的线：同一根轴上一正一反地绕，同意了就在中间打一个结
  function knotSvg(q, A, B, R) {
    var W = 300, H = 40, mid = H / 2, dA = '', dB = '', loops = 3.5;
    for (var i = 0; i <= 120; i++) {
      var t = i / 120, env = Math.sin(Math.PI * t), x = 6 + t * (W - 12);
      var ra = (3 + A.loss / 100 * 12) * env, rb = (3 + B.loss / 100 * 12) * env, a = t * loops * 6.283;
      dA += (i ? 'L' : 'M') + x.toFixed(1) + ' ' + (mid + ra * Math.sin(a)).toFixed(1);
      dB += (i ? 'L' : 'M') + x.toFixed(1) + ' ' + (mid - rb * Math.sin(a)).toFixed(1);
    }
    var tie = q.agreed ? '<circle cx="' + W / 2 + '" cy="' + mid + '" r="4" fill="#0d0c0b" stroke="#c4b08e" stroke-width="1.2"/><circle cx="' + W / 2 + '" cy="' + mid + '" r="1.4" fill="#c4b08e"/>' : '';
    var op = q.ready ? 1 : 0.4;
    return '<svg class="knot" viewBox="0 0 ' + W + ' ' + H + '" preserveAspectRatio="none" aria-hidden="true" style="opacity:' + op + '">' +
      '<path d="' + dA + '" fill="none" stroke="#a08b6b" stroke-width="1.2"/><path d="' + dB + '" fill="none" stroke="#dcd6cc" stroke-width="1" opacity=".7"/>' + tie + '</svg>';
  }
